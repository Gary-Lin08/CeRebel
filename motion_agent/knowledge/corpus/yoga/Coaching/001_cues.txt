================================================================================
DOWNWARD-FACING DOG
================================================================================

cue_dog_length_over_heels
  fault_id: fault_dog_lumbar_sag, fault_dog_rounded_spine
  pose_id: pose_down_dog
  phase: hold_anchor
  cue_text: "Long spine first; heels toward the floor only if they go without losing length."
  breath_note: Exhale soften knees; inhale lift sit bones.
  authority: B

cue_dog_push_floor
  fault_id: fault_dog_shoulder_elevation
  pose_id: pose_down_dog
  phase: hold_anchor
  cue_text: "Push the mat forward with your hands; shoulder blades slide toward your hips."
  breath_note: Steady nasal breath; avoid holding breath in shoulders.
  authority: B

cue_dog_soft_knees
  fault_id: fault_dog_locked_knees, fault_dog_rounded_spine
  pose_id: pose_down_dog
  phase: entry, hold_anchor
  cue_text: "Bend knees enough that your back can lengthen like a hammock tied high."
  breath_note: One breath per pedaling foot if using heel lift variation.
  authority: B

cue_dog_hand_width
  fault_id: fault_dog_narrow_base
  pose_id: pose_down_dog
  phase: entry
  cue_text: "Hands shoulder-width; outer edges of hands parallel to mat edge."
  breath_note: Set hands on exhale before stepping back.
  authority: B
